provider "aws" {
  region = "us-east-1"
}

resource "aws_vpc" "pathnex_vpc" {
  cidr_block = "10.0.0.0/16"
}

resource "aws_subnet" "pathnex_subnet" {
  vpc_id                  = aws_vpc.pathnex_vpc.id
  cidr_block              = "10.0.1.0/24"
  map_public_ip_on_launch = true
}

resource "aws_internet_gateway" "pathnex_igw" {
  vpc_id = aws_vpc.pathnex_vpc.id
}

resource "aws_route_table" "pathnex_rt" {
  vpc_id = aws_vpc.pathnex_vpc.id

  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.pathnex_igw.id
  }
}

resource "aws_route_table_association" "pathnex_rta" {
  subnet_id      = aws_subnet.pathnex_subnet.id
  route_table_id = aws_route_table.pathnex_rt.id
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-xxxxxxxxxxxxxxxxx"
  instance_type = "c5.xlarge"
  subnet_id     = aws_subnet.pathnex_subnet.id

  tags = {
    Name = "Pathnex-EC2"
  }
}

---
- name: Install Nginx and Deploy HTML
  hosts: all
  become: yes

  tasks:

    - name: Install Nginx
      dnf:
        name: nginx
        state: present

    - name: Start Nginx
      service:
        name: nginx
        state: started
        enabled: yes

    - name: Deploy HTML file
      copy:
        dest: /usr/share/nginx/html/index.html
        content: |
          <html>
          <body>
            <h1>Welcome to Pathnex</h1>
            <p>Nginx deployed using Ansible</p>
          </body>
          </html>

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    triggers {
        cron('H 2 * * *') // every day at 2 AM
    }
    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}
